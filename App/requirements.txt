# Backend runtime dependencies. CPU-only, ~150MB installed, no torch.
#
# Inference runs on ONNX Runtime rather than PyTorch. This is not a preference --
# the aarch64 torch wheels are built with SIMD kernels targeting Neoverse-class
# cores, and a Raspberry Pi 5 (Cortex-A76) has neither SVE nor BF16, so
# libtorch_cpu.so dies with SIGILL the instant it loads. onnxruntime ships
# baseline ARMv8 builds, and measured 3.4x faster than torch on this model.
#
# Training still uses torch -- see the root requirements.txt and
# FineTuning/export_onnx.py, which converts the trained checkpoint to model.onnx.

fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.9,<3.0
slowapi>=0.1.9

# >=1.17 because the exported graph is opset 18 (torch emits a fused
# LayerNormalization node that cannot be lowered to an earlier opset).
onnxruntime>=1.17

# tokenizer.json is a self-contained fast-tokenizer definition, so we load it
# with `tokenizers` directly instead of pulling in all of transformers -- which
# would drag torch back onto the Pi as a dependency.
tokenizers>=0.21
numpy>=1.24
